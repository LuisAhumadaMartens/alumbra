cmake_minimum_required(VERSION 3.28)
project(AlumbraConsumer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Alumbra CONFIG REQUIRED PATHS "${ALUMBRA_INSTALL_PREFIX}" NO_DEFAULT_PATH)
foreach(_target IN ITEMS umbra::base penumbra::interface)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "Installed package is missing ${_target}")
  endif()
endforeach()
if(ALUMBRA_EXPECT_PORT STREQUAL "desktop" AND NOT TARGET penumbra::desktop)
  message(FATAL_ERROR "Installed package is missing penumbra::desktop")
elseif(NOT ALUMBRA_EXPECT_PORT STREQUAL "desktop" AND TARGET penumbra::desktop)
  message(FATAL_ERROR "penumbra::desktop must not be exported without the port")
endif()
if(ALUMBRA_EXPECT_UI AND NOT TARGET umbra::ui)
  message(FATAL_ERROR "The opt-in UI target was not exported")
elseif(NOT ALUMBRA_EXPECT_UI AND TARGET umbra::ui)
  message(FATAL_ERROR "The UI target must be opt-in")
endif()

foreach(_target IN ITEMS umbra::base umbra::ui penumbra::desktop)
  if(TARGET ${_target})
    get_target_property(_dependencies ${_target} INTERFACE_LINK_LIBRARIES)
    if(NOT "penumbra::interface" IN_LIST _dependencies)
      message(FATAL_ERROR "${_target} must link penumbra::interface")
    endif()
    foreach(_dependency IN LISTS _dependencies)
      if(NOT _dependency MATCHES
          "^(penumbra::interface|\\$<LINK_ONLY:(absl::[a-z0-9_]+|Threads::Threads)>)$")
        message(FATAL_ERROR
          "${_target} must link only penumbra::interface, plus Abseil and "
          "Threads::Threads as link-only dependencies, not ${_dependency}")
      endif()
    endforeach()
  endif()
endforeach()

set(_package_targets umbra::base penumbra::interface)
if(TARGET penumbra::desktop)
  list(APPEND _package_targets penumbra::desktop)
endif()
if(ALUMBRA_EXPECT_UI)
  list(APPEND _package_targets umbra::ui)
endif()

set(_include_dirs "")
foreach(_target IN LISTS _package_targets)
  get_target_property(_dirs ${_target} INTERFACE_INCLUDE_DIRECTORIES)
  if(_dirs)
    list(APPEND _include_dirs ${_dirs})
  endif()
endforeach()
list(REMOVE_DUPLICATES _include_dirs)

if(NOT EXISTS "${ALUMBRA_LAYERING_ALLOWLIST}")
  message(FATAL_ERROR "ALUMBRA_LAYERING_ALLOWLIST must name tools/layering/allowlist.json")
endif()
file(READ "${ALUMBRA_LAYERING_ALLOWLIST}" _allowlist)
set(_standard_headers "")
string(JSON _entry_count LENGTH "${_allowlist}" includes entries)
math(EXPR _entry_last "${_entry_count} - 1")
foreach(_entry RANGE ${_entry_last})
  string(JSON _entry_id GET "${_allowlist}" includes entries ${_entry} id)
  if(_entry_id MATCHES "^(cxx|c)-standard-library$")
    string(JSON _name_count LENGTH "${_allowlist}" includes entries ${_entry} names)
    math(EXPR _name_last "${_name_count} - 1")
    foreach(_name RANGE ${_name_last})
      string(JSON _standard_header GET "${_allowlist}"
        includes entries ${_entry} names ${_name})
      list(APPEND _standard_headers "${_standard_header}")
    endforeach()
  endif()
endforeach()

set(_header_violations "")
set(_header_sources "")
foreach(_include_dir IN LISTS _include_dirs)
  file(GLOB_RECURSE _headers RELATIVE "${_include_dir}"
    "${_include_dir}/*.h" "${_include_dir}/*.hpp"
    "${_include_dir}/*.hh" "${_include_dir}/*.hxx")
  foreach(_header IN LISTS _headers)
    file(STRINGS "${_include_dir}/${_header}" _directives
      REGEX "^[ \t]*#[ \t]*include")
    foreach(_directive IN LISTS _directives)
      string(REGEX MATCH "include[ \t]*[<\"]([^>\"]+)[>\"]" _match "${_directive}")
      set(_included "${CMAKE_MATCH_1}")
      if(_included IN_LIST _standard_headers)
        continue()
      endif()
      if(_included MATCHES "^(umbra|penumbra)/")
        set(_installed FALSE)
        foreach(_candidate_dir IN LISTS _include_dirs)
          if(EXISTS "${_candidate_dir}/${_included}")
            set(_installed TRUE)
          endif()
        endforeach()
        if(NOT _installed)
          list(APPEND _header_violations
            "${_header} includes ${_included}, which is not installed")
        endif()
        continue()
      endif()
      list(APPEND _header_violations
        "${_header} includes ${_included}, which is neither the C++ standard library nor an installed Alumbra header")
    endforeach()
    string(MAKE_C_IDENTIFIER "${_header}" _name)
    set(_source "${CMAKE_CURRENT_BINARY_DIR}/public_headers/${_name}.cc")
    file(CONFIGURE OUTPUT "${_source}"
      CONTENT "#include \"@_header@\"\n\nstatic_assert(__cplusplus == 202002L);\n"
      @ONLY)
    list(APPEND _header_sources "${_source}")
  endforeach()
endforeach()
if(_header_violations)
  list(JOIN _header_violations "\n  " _header_violations)
  message(FATAL_ERROR
    "Installed headers may include only the C++ standard library and other "
    "installed Alumbra headers:\n  ${_header_violations}")
endif()
if(NOT _header_sources)
  message(FATAL_ERROR "No installed public headers found in ${_include_dirs}")
endif()
add_library(alumbra_public_headers OBJECT ${_header_sources})
target_link_libraries(alumbra_public_headers PRIVATE ${_package_targets})

add_executable(alumbra_consumer main.cc)
target_link_libraries(alumbra_consumer PRIVATE umbra::base)
if(TARGET penumbra::desktop)
  target_link_libraries(alumbra_consumer PRIVATE penumbra::desktop)
endif()
if(ALUMBRA_EXPECT_UI)
  target_link_libraries(alumbra_consumer PRIVATE umbra::ui)
endif()
enable_testing()
add_test(NAME installed_link_probe COMMAND alumbra_consumer)
